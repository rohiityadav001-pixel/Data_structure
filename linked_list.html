LINKED LIST OPERATIONS

1. Traversing a Linked List

Program to display all elements of a linked list.

#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *next;
};

int main()
{
    struct node *head, *second, *third, *temp;

    head = malloc(sizeof(struct node));
    second = malloc(sizeof(struct node));
    third = malloc(sizeof(struct node));

    head->data = 10;
    head->next = second;

    second->data = 20;
    second->next = third;

    third->data = 30;
    third->next = NULL;

    temp = head;

    while(temp != NULL)
    {
        printf("%d ", temp->data);
        temp = temp->next;
    }

    return 0;
}
Output:

10 20 30

2. Searching in Linked List

Program to search an element in a linked list.

#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *next;
};

int main()
{
    struct node *head, *temp;
    int key = 20, found = 0;

    head = malloc(sizeof(struct node));
    head->data = 10;

    head->next = malloc(sizeof(struct node));
    head->next->data = 20;

    head->next->next = malloc(sizeof(struct node));
    head->next->next->data = 30;
    head->next->next->next = NULL;

    temp = head;

    while(temp != NULL)
    {
        if(temp->data == key)
        {
            printf("Element Found");
            found = 1;
            break;
        }

        temp = temp->next;
    }

    if(found == 0)
        printf("Element Not Found");

    return 0;
}
Output:

Element Found

3. Insertion in Linked List

Program to insert a node at the beginning.

#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *next;
};

int main()
{
    struct node *head, *newnode, *temp;

    head = malloc(sizeof(struct node));
    head->data = 20;
    head->next = NULL;

    newnode = malloc(sizeof(struct node));

    newnode->data = 10;
    newnode->next = head;

    head = newnode;

    temp = head;

    printf("Linked List: ");

    while(temp != NULL)
    {
        printf("%d ", temp->data);
        temp = temp->next;
    }

    return 0;
}
Output:

Linked List: 10 20

4. Deletion in Linked List

Program to delete a node from the beginning.

#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *next;
};

int main()
{
    struct node *head, *temp;

    head = malloc(sizeof(struct node));
    head->data = 10;

    head->next = malloc(sizeof(struct node));
    head->next->data = 20;
    head->next->next = NULL;

    temp = head;

    head = head->next;

    free(temp);

    temp = head;

    printf("Linked List: ");

    while(temp != NULL)
    {
        printf("%d ", temp->data);
        temp = temp->next;
    }

    return 0;
}
Output:

Linked List: 20

Important Points

  • Traversal: Display all nodes.
  • Searching: Find a particular element.
  • Insertion: Add a new node.
  • Deletion: Remove a node.
