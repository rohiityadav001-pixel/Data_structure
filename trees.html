1. Insertion in Binary Tree

This program inserts a new node at the first available position using level order.

C Program

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node* insert(struct Node* root, int value) {
    struct Node* newNode =
        (struct Node*)malloc(sizeof(struct Node));

    newNode->data = value;
    newNode->left = NULL;
    newNode->right = NULL;

    if (root == NULL)
        return newNode;

    struct Node* queue[100];
    int front = 0, rear = 0;

    queue[rear++] = root;

    while (front < rear) {
        struct Node* temp = queue[front++];

        if (temp->left == NULL) {
            temp->left = newNode;
            return root;
        }
        else
            queue[rear++] = temp->left;

        if (temp->right == NULL) {
            temp->right = newNode;
            return root;
        }
        else
            queue[rear++] = temp->right;
    }

    return root;
}

void display(struct Node* root) {
    if (root == NULL)
        return;

    struct Node* queue[100];
    int front = 0, rear = 0;

    queue[rear++] = root;

    while (front < rear) {
        struct Node* temp = queue[front++];
        printf("%d ", temp->data);

        if (temp->left != NULL)
            queue[rear++] = temp->left;

        if (temp->right != NULL)
            queue[rear++] = temp->right;
    }
}

int main() {
    struct Node* root = NULL;

    root = insert(root, 10);
    root = insert(root, 20);
    root = insert(root, 30);
    root = insert(root, 40);

    printf("Binary Tree: ");
    display(root);

    return 0;
}

Output

Binary Tree: 10 20 30 40

2. Deletion in Binary Tree

This program deletes a node by replacing it with the deepest rightmost node.

C Program

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

void deleteNode(struct Node* root, int key) {
    if (root == NULL)
        return;

    if (root->left == NULL && root->right == NULL) {
        if (root->data == key)
            root->data = -1;
        return;
    }

    struct Node* queue[100];
    struct Node* temp;
    struct Node* keyNode = NULL;
    int front = 0, rear = 0;

    queue[rear++] = root;

    while (front < rear) {
        temp = queue[front++];

        if (temp->data == key)
            keyNode = temp;

        if (temp->left != NULL)
            queue[rear++] = temp->left;

        if (temp->right != NULL)
            queue[rear++] = temp->right;
    }

    if (keyNode != NULL) {
        int deepestData = temp->data;
        keyNode->data = deepestData;

        front = 0;
        rear = 0;
        queue[rear++] = root;

        while (front < rear) {
            temp = queue[front++];

            if (temp->left != NULL) {
                if (temp->left == NULL)
                    break;

                if (temp->left->left == NULL &&
                    temp->left->right == NULL &&
                    temp->left->data == deepestData) {
                    free(temp->left);
                    temp->left = NULL;
                    return;
                }
                queue[rear++] = temp->left;
            }

            if (temp->right != NULL) {
                if (temp->right->left == NULL &&
                    temp->right->right == NULL &&
                    temp->right->data == deepestData) {
                    free(temp->right);
                    temp->right = NULL;
                    return;
                }
                queue[rear++] = temp->right;
            }
        }
    }
}

void display(struct Node* root) {
    if (root == NULL)
        return;

    struct Node* queue[100];
    int front = 0, rear = 0;

    queue[rear++] = root;

    while (front < rear) {
        struct Node* temp = queue[front++];
        printf("%d ", temp->data);

        if (temp->left != NULL)
            queue[rear++] = temp->left;

        if (temp->right != NULL)
            queue[rear++] = temp->right;
    }
}

struct Node* createNode(int value) {
    struct Node* node = malloc(sizeof(struct Node));
    node->data = value;
    node->left = node->right = NULL;
    return node;
}

int main() {
    struct Node* root = createNode(10);
    root->left = createNode(20);
    root->right = createNode(30);
    root->left->left = createNode(40);
    root->left->right = createNode(50);

    printf("Before Deletion: ");
    display(root);

    deleteNode(root, 20);

    printf("\nAfter Deletion: ");
    display(root);

    return 0;
}

Output

Before Deletion: 10 20 30 40 50
After Deletion: 10 50 30 40

3. Searching in Binary Tree

This program searches for a value using level-order traversal.

C Program

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

int search(struct Node* root, int key) {
    if (root == NULL)
        return 0;

    struct Node* queue[100];
    int front = 0, rear = 0;

    queue[rear++] = root;

    while (front < rear) {
        struct Node* temp = queue[front++];

        if (temp->data == key)
            return 1;

        if (temp->left != NULL)
            queue[rear++] = temp->left;

        if (temp->right != NULL)
            queue[rear++] = temp->right;
    }

    return 0;
}

struct Node* createNode(int value) {
    struct Node* node = malloc(sizeof(struct Node));
    node->data = value;
    node->left = node->right = NULL;
    return node;
}

int main() {
    struct Node* root = createNode(10);
    root->left = createNode(20);
    root->right = createNode(30);
    root->left->left = createNode(40);
    root->left->right = createNode(50);

    int key = 40;

    if (search(root, key))
        printf("%d is Found in Binary Tree", key);
    else
        printf("%d is Not Found in Binary Tree", key);

    return 0;
}

Output

40 is Found in Binary Tree
