STACK OPERATIONS

1. Stack Traversal

Program to display all elements of a stack.

#include <stdio.h>

int main()
{
    int stack[5] = {10, 20, 30, 40, 50};
    int top = 4, i;

    printf("Stack Elements:\n");

    for(i = top; i >= 0; i--)
    {
        printf("%d\n", stack[i]);
    }

    return 0;
}
Output: 
Stack Elements:
50
40
30
20
10

2. Stack Insertion (PUSH)

Program to insert an element into the stack.

#include <stdio.h>

int main()
{
    int stack[5] = {10, 20, 30};
    int top = 2;
    int item = 40;

    if(top == 4)
    {
        printf("Stack Overflow");
    }
    else
    {
        top++;
        stack[top] = item;

        printf("Element Inserted\n");
        printf("Stack Elements:\n");

        for(int i = top; i >= 0; i--)
        {
            printf("%d\n", stack[i]);
        }
    }

    return 0;
}
Output: 
Element Inserted
Stack Elements:
40
30
20
10

3. Stack Deletion (POP)

Program to delete an element from the stack.

#include <stdio.h>

int main()
{
    int stack[5] = {10, 20, 30, 40};
    int top = 3;

    if(top == -1)
    {
        printf("Stack Underflow");
    }
    else
    {
        printf("Deleted Element = %d\n", stack[top]);
        top--;

        printf("Stack Elements:\n");

        for(int i = top; i >= 0; i--)
        {
            printf("%d\n", stack[i]);
        }
    }

    return 0;
}
Output: 
Deleted Element = 40
Stack Elements:
30
20
10

Stack Operations

  • Push: Insertion of an element into the stack.
  • Pop: Deletion of the top element.
  • Traversal: Displaying all stack elements.
