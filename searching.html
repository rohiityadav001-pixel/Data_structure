SEARCHING PROGRAMS

1. Linear Search

Program to search an element using Linear Search.

#include <stdio.h>

int main()
{
    int a[5] = {10, 20, 30, 40, 50};
    int key = 30;
    int i, found = 0;

    for(i = 0; i < 5; i++)
    {
        if(a[i] == key)
        {
            printf("Element Found at position %d", i+1);
            found = 1;
            break;
        }
    }

    if(found == 0)
        printf("Element Not Found");

    return 0;
}
Output:

Element Found at position 3

2. Binary Search

Program to search an element using Binary Search.

#include <stdio.h>

int main()
{
    int a[5] = {10, 20, 30, 40, 50};
    int key = 30;

    int low = 0, high = 4;
    int mid, found = 0;

    while(low <= high)
    {
        mid = (low + high) / 2;

        if(a[mid] == key)
        {
            printf("Element Found at position %d", mid+1);
            found = 1;
            break;
        }

        else if(a[mid] < key)
        {
            low = mid + 1;
        }

        else
        {
            high = mid - 1;
        }
    }

    if(found == 0)
        printf("Element Not Found");

    return 0;
}
Output:

Element Found at position 3

Difference Between Linear and Binary Search

Linear Search Binary Search
Checks elements one by one. Divides the array into halves.
Works on sorted and unsorted arrays. Requires a sorted array.
Time: O(n) Time: O(log n)
