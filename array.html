Data Structures Programs

1. Array Traversal

Program to display all elements of an array.

#include <stdio.h>

int main()
{
    int a[5] = {10, 20, 30, 40, 50};

    for(int i = 0; i < 5; i++)
    {
        printf("%d ", a[i]);
    }

    return 0;
}
Output:

10 20 30 40 50

2. Stack Using Array

Program to perform push operation.

#include <stdio.h>

int main()
{
    int stack[5], top = -1;

    top++;
    stack[top] = 10;

    printf("Element: %d", stack[top]);

    return 0;
}
Output:

Element: 10

3. Linear Search

Program to search an element in an array.

#include <stdio.h>

int main()
{
    int a[5] = {10, 20, 30, 40, 50};
    int key = 30;

    for(int i = 0; i < 5; i++)
    {
        if(a[i] == key)
        {
            printf("Element Found");
            return 0;
        }
    }

    printf("Element Not Found");

    return 0;
}
Output:

Element Found

4. Insertion in array

Program to insert an element in an array.

#include <stdio.h>

int main()
{
    int a[10] = {10, 20, 30, 40};
    int n = 4, pos = 3, item = 25;
    int i;

    for(i = n; i >= pos; i--)
    {
        a[i] = a[i-1];
    }

    a[pos-1] = item;
    n++;

    printf("Array after insertion:\n");

    for(i = 0; i < n; i++)
    {
        printf("%d ", a[i]);
    }

    return 0;
}
Output:

Array after insertion: 10 20 25 30 40

5. Deletion in Array

Program to delete an element in an array. 

 
            #include  <stdio.h>

int main()
{
    int a[10] = {10, 20, 30, 40, 50};
    int n = 5, pos = 3;
    int i;

    for(i = pos-1; i < n-1; i++)
    {
        a[i] = a[i+1];
    }

    n--;

    printf("Array after deletion:\n");

    for(i = 0; i < n; i++)
    {
        printf("%d ", a[i]);
    }

    return 0;
}
Output

Array after deletion: 10 20 40 50
